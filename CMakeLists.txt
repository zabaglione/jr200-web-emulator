cmake_minimum_required(VERSION 3.20)
project(jr200_web VERSION 0.0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(JR200_SANITIZE "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
add_library(jr200_cjr STATIC src/tape/cjr.cpp)
target_include_directories(jr200_cjr PUBLIC include)
if(MSVC)
  target_compile_options(jr200_cjr PRIVATE /W4)
else()
  target_compile_options(jr200_cjr PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
if(JR200_SANITIZE AND NOT EMSCRIPTEN AND NOT MSVC)
  target_compile_options(jr200_cjr PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(jr200_cjr PUBLIC -fsanitize=address,undefined)
endif()
if(EMSCRIPTEN)
  add_executable(jr200_codec src/wasm/api.cpp)
  target_link_libraries(jr200_codec PRIVATE jr200_cjr)
  set_target_properties(jr200_codec PROPERTIES SUFFIX ".mjs" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/web")
  target_link_options(jr200_codec PRIVATE
    "--no-entry" "-sMODULARIZE=1" "-sEXPORT_NAME=createJR200Codec"
    "-sEXPORT_ES6=1" "-sENVIRONMENT=web,node" "-sFILESYSTEM=0"
    "-sALLOW_MEMORY_GROWTH=0" "-sINITIAL_MEMORY=16777216"
    "-sEXPORTED_FUNCTIONS=['_jr200_codec_api_version','_jr200_input_ptr','_jr200_output_ptr','_jr200_capacity','_jr200_output_size','_jr200_error_offset','_jr200_error_message','_jr200_inspect','_jr200_summary_field','_jr200_name_ptr','_jr200_encode']")
else()
  include(CTest)
  add_executable(cjrtool tools/cjrtool.cpp)
  target_link_libraries(cjrtool PRIVATE jr200_cjr)
  add_executable(test_cjr tests/test_cjr.cpp src/wasm/api.cpp)
  target_link_libraries(test_cjr PRIVATE jr200_cjr)
  add_test(NAME cjr_native COMMAND test_cjr)
  find_package(Python3 COMPONENTS Interpreter REQUIRED)
  add_test(NAME license_guard COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/check_distribution.py)
  add_test(NAME bootstrap_unit COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_bootstrap.py)
endif()
